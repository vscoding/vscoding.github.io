Access Control

基于 Lua 的访问控制设计

-- 访问控制（ACL）规则匹配控制器 -- 提供基于规则列表的匹配与检查：黑名单（命中即拒绝）、白名单（命中即放行） local _M = {} local obj_utils = require("lua_libs.commons.obj_utils") local string_utils = require("lua_libs.commons.string_utils") local ac_utils = require("lua_libs.http_req.access_control.ac_utils") -- 判断单条规则是否命中当前请求 -- 规则结构示例：{field="host", operator="equals", value="example.com", effect="deny"} -- @param ctx 请求上下文，rule.field 对应的字段值取自 ctx -- @param rule 规则本身 -- @param manager 可选的规则管理器，默认为 rules_manager -- @return boolean 是否命中 local function match_rule(ctx, rule, manager) -- 取出参与匹配的三要素：字段值（来自 ctx）、操作符、规则值 local field_val, operator, value = ctx[rule.field], rule.operator, rule.value -- value_func：通过回调函数计算匹配结果，函数入参为字段值 field_val if operator == "value_func" then -- 获取计算匹配结果的函数 local value_func, err = ac_utils.get_value_func(rule, manager) if not value_func then ngx.log(ngx.ERR, "failed to get value_func rule: ", err or "unknown error") return false end if not obj_utils.is_function(value_func) then ngx.log(ngx.ERR, "value_func must be a function when operator is 'value_func'") return false end -- 调用函数，函数异常或返回非布尔值时视为不命中 local ok, res = pcall(value_func, field_val) if not ok then return false end if not obj_utils.is_boolean(res) then return false end return res end -- @function：调用内置函数完成匹配，函数入参为整个请求上下文 ctx if operator == "@function" then local func, err = ac_utils.get_internal_function(rule, manager) if not func then ngx.log(ngx.ERR, "failed to get function: ", err or "unknown error") return false end if not obj_utils.is_function(func) then ngx.log(ngx.ERR, "func must be a function when operator is '@function'") return false end -- 调用函数，函数异常或返回非布尔值时视为不命中 local ok, res = pcall(func, ctx) if not ok then return false end if not obj_utils.is_boolean(res) then return false end return res end -- 通用操作符：field_val 和 value 均须为非空字符串，否则跳过该规则 local field_val_blank = string_utils.is_blank(field_val) local value_blank = string_utils.is_blank(value) if field_val_blank or value_blank then ngx.log( ngx.WARN, "ACL rule skipped because field_val or value is blank/non-string: operator=", tostring(operator), ", field=", tostring(rule.field), ", field_val_type=", type(field_val), ", value_type=", type(value) ) return false end -- 支持的通用操作符：equals / not_equals / prefix / suffix / contains / regex / in if operator == "equals" then return field_val == value elseif operator == "not_equals" then return field_val ~= value elseif operator == "prefix" then return field_val:sub(1, #value) == value elseif operator == "suffix" then return field_val:sub(-#value) == value elseif operator == "contains" then return field_val:find(value, 1, true) ~= nil elseif operator == "regex" then return ngx.re.find(field_val, value, "jo") ~= nil elseif operator == "in" then -- 规则值以英文逗号分隔，字段值命中任意一项即匹配 for v in value:gmatch("[^,]+") do if field_val == v then return true end end return false else -- 未知操作符：视为不命中 return false end end -- 核心检查函数 -- 黑名单模式：命中 effect="deny" 的规则即拒绝，未命中默认放行 -- 白名单模式：命中 effect="allow" 的规则即放行，未命中默认拒绝 -- @param ctx 请求上下文 -- @param rules 规则列表，规则示例 {field="host", operator="equals", value="example.com", effect="deny"} -- @param mode 检查模式："whitelist" 或 "blacklist"，默认为 "blacklist" -- @param manager 可选的规则管理器，默认为 rules_manager -- @return allowed boolean 是否允许通过；rule 命中的规则（未命中时为 nil） local function check_ctx(ctx, rules, mode, manager) mode = mode or "blacklist" for _, rule in ipairs(rules) do if match_rule(ctx, rule, manager) then -- 规则命中：黑名单命中 deny 即拒绝，白名单命中 allow 即放行 if mode == "blacklist" and rule.effect == "deny" then return false, rule elseif mode == "whitelist" and rule.effect == "allow" then return true, rule end end end -- 规则列表为空时记录告警 if #rules == 0 then ngx.log(ngx.WARN, "No rules provided for access control check in mode: ", mode) end -- 没有任何规则命中：黑名单默认放行，白名单默认拒绝 if mode == "blacklist" then return true, nil else return false, nil end end -- 模块方法：按指定模式执行访问控制检查 -- @param ctx 请求上下文 -- @param rules 规则列表 -- @param mode 检查模式："whitelist" 或 "blacklist" -- @param manager 可选的规则管理器，默认为 rules_manager -- @return allowed boolean 是否允许通过；rule 命中的规则（未命中时为 nil） function _M.check_ctx(ctx, rules, mode, manager) return check_ctx(ctx, rules, mode, manager) end -- 模块方法：白名单检查（命中 effect="allow" 的规则即放行，未命中默认拒绝） -- @param ctx 请求上下文 -- @param rules 规则列表 -- @param manager 可选的规则管理器，默认为 rules_manager -- @return allowed boolean 是否允许通过；rule 命中的规则（未命中时为 nil） function _M.check_ctx_whitelist(ctx, rules, manager) return check_ctx(ctx, rules, "whitelist", manager) end -- 模块方法：黑名单检查（命中 effect="deny" 的规则即拒绝，未命中默认放行） -- @param ctx 请求上下文 -- @param rules 规则列表 -- @param manager 可选的规则管理器，默认为 rules_manager -- @return allowed boolean 是否允许通过；rule 命中的规则（未命中时为 nil） function _M.check_ctx_blacklist(ctx, rules, manager) return check_ctx(ctx, rules, "blacklist", manager) end return _M
