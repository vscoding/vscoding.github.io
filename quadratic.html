InsertSort.kt

源码： sort/impl/InsertSort.kt

/** * InsertSort 插入排序 */ class InsertSort<T : Comparable<T>> : Sort<T> { override fun sort(arr: Array<T>) { if (arr.isEmpty() || arr.size == 1) return val n = arr.size for (i in 1 until n) { for (j in i downTo 1) { if (arr[j] < arr[j - 1]) { arr.swap(j, j - 1) } } } } }

/** * SelectSort 选择排序 */ class SelectSort<T : Comparable<T>> : Sort<T> { override fun sort(arr: Array<T>) { if (arr.isEmpty() || arr.size == 1) return val n = arr.size for (i in 0 until n) { for (j in i + 1 until n) { if (arr[j] < arr[i]) { arr.swap(i, j) } } } } }
